# language: CMake, file: CMakeLists.txt, runtime: CMake 3.21+, target: Windows 11/MSVC/Qt 6.10
cmake_minimum_required(VERSION 3.21)
project(tds_plus_qt LANGUAGES CXX RC)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Quick QuickControls2 ShaderTools)

set(APP_SOURCES qt/main.cpp qt/control_model.cpp qt/control_model.hpp tds_plus.cpp backend_api.hpp)
set(UI_ASSETS qml/Main.qml qml/AmbientBackground.qml qml/TitleBar.qml qt/tds_plus.ico
    assets/fonts/GeistMono-Medium.ttf assets/fonts/GeistMono-SemiBold.ttf
    assets/fonts/OFL.txt assets/fonts/SOURCE.txt)
qt_add_executable(tds_plus_qt WIN32 ${APP_SOURCES} qt/app.rc)
qt_add_resources(tds_plus_qt ui_resources PREFIX "/" FILES ${UI_ASSETS})
qt_add_shaders(tds_plus_qt ambient_shader PREFIX "/shaders" OUTPUTS ambient.frag.qsb FILES shaders/ambient.frag)
target_compile_definitions(tds_plus_qt PRIVATE TDS_PLUS_QT QT_NO_KEYWORDS NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_link_libraries(tds_plus_qt PRIVATE Qt6::Quick Qt6::QuickControls2 psapi ws2_32 dwmapi)
if(MSVC)
    target_compile_options(tds_plus_qt PRIVATE /utf-8 /W4)
endif()

option(TDS_BUILD_UI_TEST "Build the isolated desktop UI smoke test" OFF)
if(TDS_BUILD_UI_TEST)
    enable_testing()
    add_executable(tds_plus_chain_state_test tests/chain_state_test.cpp)
    set_target_properties(tds_plus_chain_state_test PROPERTIES AUTOMOC OFF)
    add_test(NAME chain_state COMMAND tds_plus_chain_state_test)
    add_executable(tds_plus_frame_zone_test tests/frame_zone_test.cpp)
    set_target_properties(tds_plus_frame_zone_test PROPERTIES AUTOMOC OFF)
    if(MSVC)
        target_compile_options(tds_plus_frame_zone_test PRIVATE /utf-8 /W4)
    endif()
    add_test(NAME frame_zone COMMAND tds_plus_frame_zone_test)
    add_executable(tds_plus_game_state_test tests/game_state_test.cpp)
    set_target_properties(tds_plus_game_state_test PROPERTIES AUTOMOC OFF)
    target_compile_definitions(tds_plus_game_state_test PRIVATE TDS_PLUS_QT NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(tds_plus_game_state_test PRIVATE psapi ws2_32 user32)
    if(MSVC)
        target_compile_options(tds_plus_game_state_test PRIVATE /utf-8 /W4)
    endif()
    add_test(NAME game_state COMMAND tds_plus_game_state_test)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(tds_plus_ui_test WIN32 ${APP_SOURCES} qt/smoke_test.hpp)
    qt_add_resources(tds_plus_ui_test test_ui_resources PREFIX "/" FILES ${UI_ASSETS})
    qt_add_shaders(tds_plus_ui_test test_ambient_shader PREFIX "/shaders" OUTPUTS ambient.frag.qsb FILES shaders/ambient.frag)
    target_compile_definitions(tds_plus_ui_test PRIVATE TDS_PLUS_QT TDS_UI_SMOKE QT_NO_KEYWORDS NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(tds_plus_ui_test PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Test psapi ws2_32 dwmapi)
    if(MSVC)
        target_compile_options(tds_plus_ui_test PRIVATE /utf-8 /W4)
    endif()
endif()
